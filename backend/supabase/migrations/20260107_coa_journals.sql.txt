-- Chart of Accounts and Advanced Journal Entries
-- Part 6: CoA structure and Journal Line Items

-- Chart of Accounts Categories
CREATE TABLE IF NOT EXISTS accounting_account_categories (
    id SERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL, -- ASSETS, LIABILITIES, EQUITY, REVENUE, EXPENSES
    normal_balance VARCHAR(10) CHECK (normal_balance IN ('DEBIT', 'CREDIT')),
    code_range_start INTEGER,
    code_range_end INTEGER
);

-- Chart of Accounts
CREATE TABLE IF NOT EXISTS accounting_chart_of_accounts (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    branch_id INTEGER REFERENCES branches(id),
    category_id INTEGER REFERENCES accounting_account_categories(id),
    account_code VARCHAR(20) UNIQUE NOT NULL,
    account_name VARCHAR(255) NOT NULL,
    description TEXT,
    is_active BOOLEAN DEFAULT TRUE,
    parent_account_id UUID REFERENCES accounting_chart_of_accounts(id),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Detailed Journal Line Items (Double Entry)
CREATE TABLE IF NOT EXISTS accounting_journal_items (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    journal_entry_id UUID NOT NULL REFERENCES accounting_journal_entries(id) ON DELETE CASCADE,
    account_id UUID NOT NULL REFERENCES accounting_chart_of_accounts(id),
    description TEXT,
    debit DECIMAL(12,2) DEFAULT 0,
    credit DECIMAL(12,2) DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    CONSTRAINT balanced_entry CHECK (debit >= 0 AND credit >= 0 AND (debit > 0 OR credit > 0))
);

-- Views for Financial Reporting
CREATE OR REPLACE VIEW trial_balance AS
SELECT 
    coa.account_code,
    coa.account_name,
    cat.name as category,
    SUM(ji.debit) as total_debit,
    SUM(ji.credit) as total_credit,
    CASE 
        WHEN cat.normal_balance = 'DEBIT' THEN SUM(ji.debit) - SUM(ji.credit)
        ELSE SUM(ji.credit) - SUM(ji.debit)
    END as net_balance
FROM accounting_chart_of_accounts coa
JOIN accounting_account_categories cat ON coa.category_id = cat.id
LEFT JOIN accounting_journal_items ji ON coa.id = ji.account_id
GROUP BY coa.id, coa.account_code, coa.account_name, cat.name, cat.normal_balance;

-- Indexes
CREATE INDEX IF NOT EXISTS idx_coa_category ON accounting_chart_of_accounts(category_id);
CREATE INDEX IF NOT EXISTS idx_journal_items_account ON accounting_journal_items(account_id);
CREATE INDEX IF NOT EXISTS idx_journal_items_entry ON accounting_journal_items(journal_entry_id);
