-- Invoicing, Quotations and Auditor Integration
-- Part 5: Line Items, Quotations, and Audit Logs

-- Invoice Line Items
CREATE TABLE IF NOT EXISTS customer_invoice_items (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  invoice_id UUID NOT NULL REFERENCES customer_invoices(id) ON DELETE CASCADE,
  item_description TEXT NOT NULL,
  quantity DECIMAL(10,2) NOT NULL,
  unit_price DECIMAL(12,2) NOT NULL,
  tax_rate DECIMAL(5,2) DEFAULT 16.00, -- Default VAT
  tax_amount DECIMAL(12,2) GENERATED ALWAYS AS (quantity * unit_price * tax_rate / 100) STORED,
  total_amount DECIMAL(12,2) GENERATED ALWAYS AS (quantity * unit_price * (1 + tax_rate / 100)) STORED,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Quotations table
CREATE TABLE IF NOT EXISTS quotations (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  branch_id INTEGER NOT NULL REFERENCES branches(id),
  customer_id UUID REFERENCES guests(id),
  quotation_number VARCHAR(50) UNIQUE NOT NULL,
  quotation_date DATE NOT NULL,
  expiry_date DATE NOT NULL,
  subtotal DECIMAL(12,2) NOT NULL,
  tax_total DECIMAL(12,2) NOT NULL,
  grand_total DECIMAL(12,2) NOT NULL,
  status VARCHAR(20) DEFAULT 'draft' CHECK (status IN ('draft', 'sent', 'accepted', 'rejected', 'expired', 'converted')),
  terms_and_conditions TEXT,
  notes TEXT,
  created_by UUID REFERENCES staff(id),
  converted_to_invoice_id UUID REFERENCES customer_invoices(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Quotation Line Items
CREATE TABLE IF NOT EXISTS quotation_items (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  quotation_id UUID NOT NULL REFERENCES quotations(id) ON DELETE CASCADE,
  item_description TEXT NOT NULL,
  quantity DECIMAL(10,2) NOT NULL,
  unit_price DECIMAL(12,2) NOT NULL,
  tax_rate DECIMAL(5,2) DEFAULT 16.00,
  tax_amount DECIMAL(12,2) GENERATED ALWAYS AS (quantity * unit_price * tax_rate / 100) STORED,
  total_amount DECIMAL(12,2) GENERATED ALWAYS AS (quantity * unit_price * (1 + tax_rate / 100)) STORED,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Central Audit Logs for Financial Operations
CREATE TABLE IF NOT EXISTS financial_audit_logs (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  branch_id INTEGER REFERENCES branches(id),
  user_id UUID REFERENCES staff(id),
  action_type VARCHAR(50) NOT NULL, -- 'CREATE', 'UPDATE', 'DELETE', 'APPROVE', 'REJECT', 'CLOSE_PERIOD'
  entity_type VARCHAR(50) NOT NULL, -- 'STOCK_COUNT', 'EXPENSE', 'INVOICE', 'JOURNAL_ENTRY'
  entity_id UUID NOT NULL,
  old_data JSONB,
  new_data JSONB,
  ip_address VARCHAR(45),
  user_agent TEXT,
  notes TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Approval Workflow Tracking
CREATE TABLE IF NOT EXISTS approval_requests (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  branch_id INTEGER NOT NULL REFERENCES branches(id),
  requested_by UUID NOT NULL REFERENCES staff(id),
  request_type VARCHAR(50) NOT NULL, -- 'STOCK_ADJUSTMENT', 'EXPENSE_APPROVAL', 'JOURNAL_POSTING'
  entity_id UUID NOT NULL,
  entity_type VARCHAR(50) NOT NULL,
  status VARCHAR(20) DEFAULT 'pending' CHECK (status IN ('pending', 'approved', 'rejected')),
  priority VARCHAR(10) DEFAULT 'normal' CHECK (priority IN ('low', 'normal', 'high', 'urgent')),
  comments TEXT,
  reviewed_by UUID REFERENCES staff(id),
  reviewed_at TIMESTAMP WITH TIME ZONE,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Indexes
CREATE INDEX IF NOT EXISTS idx_quotations_branch ON quotations(branch_id);
CREATE INDEX IF NOT EXISTS idx_quotations_status ON quotations(status);
CREATE INDEX IF NOT EXISTS idx_audit_logs_entity ON financial_audit_logs(entity_type, entity_id);
CREATE INDEX IF NOT EXISTS idx_audit_logs_user ON financial_audit_logs(user_id);
CREATE INDEX IF NOT EXISTS idx_approval_requests_status ON approval_requests(status);
CREATE INDEX IF NOT EXISTS idx_approval_requests_branch ON approval_requests(branch_id);

-- Trigger to log record creation in audit logs (example for stock counts)
CREATE OR REPLACE FUNCTION log_financial_action()
RETURNS TRIGGER AS $$
BEGIN
  INSERT INTO financial_audit_logs (
    branch_id, 
    user_id, 
    action_type, 
    entity_type, 
    entity_id, 
    new_data
  )
  VALUES (
    NEW.branch_id,
    auth.uid(),
    TG_OP,
    TG_TABLE_NAME,
    NEW.id,
    to_jsonb(NEW)
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Comments
COMMENT ON TABLE customer_invoice_items IS 'Detailed lines for customer invoices';
COMMENT ON TABLE quotations IS 'Quotes and estimates for customers';
COMMENT ON TABLE financial_audit_logs IS 'Centralized audit trail for all financial modifications';
COMMENT ON TABLE approval_requests IS 'Tracking system for multi-level financial approvals';
