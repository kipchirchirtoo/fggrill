-- Employee Credit & Customer Unpaid Bills
-- Part 2: Credit Management & Aging Analysis

-- Employee credit bills tracking
CREATE TABLE IF NOT EXISTS employee_credit_bills (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  employee_id UUID NOT NULL REFERENCES staff_profiles(id),
  branch_id INTEGER NOT NULL REFERENCES branches(id),
  bill_number VARCHAR(50) UNIQUE NOT NULL,
  bill_date DATE NOT NULL,
  amount DECIMAL(12,2) NOT NULL,
  paid_amount DECIMAL(12,2) DEFAULT 0,
  balance DECIMAL(12,2) GENERATED ALWAYS AS (amount - paid_amount) STORED,
  credit_limit DECIMAL(12,2),
  status VARCHAR(20) DEFAULT 'unpaid' CHECK (status IN ('unpaid', 'partial', 'paid', 'written_off')),
  description TEXT,
  due_date DATE,
  recorded_by UUID REFERENCES staff_profiles(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Payments for employee credit bills
CREATE TABLE IF NOT EXISTS employee_credit_payments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  credit_bill_id UUID NOT NULL REFERENCES employee_credit_bills(id) ON DELETE CASCADE,
  payment_date DATE NOT NULL,
  amount DECIMAL(12,2) NOT NULL,
  payment_method VARCHAR(20) CHECK (payment_method IN ('cash', 'bank', 'mpesa', 'deduction')),
  reference VARCHAR(100),
  recorded_by UUID REFERENCES staff_profiles(id),
  notes TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Customer/Guest unpaid invoices
CREATE TABLE IF NOT EXISTS customer_invoices (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  customer_id UUID REFERENCES guests(id),
  branch_id INTEGER NOT NULL REFERENCES branches(id),
  invoice_number VARCHAR(50) UNIQUE NOT NULL,
  invoice_date DATE NOT NULL,
  due_date DATE NOT NULL,
  amount DECIMAL(12,2) NOT NULL,
  paid_amount DECIMAL(12,2) DEFAULT 0,
  balance DECIMAL(12,2) GENERATED ALWAYS AS (amount - paid_amount) STORED,
  status VARCHAR(20) DEFAULT 'unpaid' CHECK (status IN ('unpaid', 'partial', 'paid', 'overdue', 'written_off')),
  aging_bucket VARCHAR(20), -- 'current', '1-30', '31-60', '61-90', '90+'
  description TEXT,
  created_by UUID REFERENCES staff_profiles(id),
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Payments for customer invoices
CREATE TABLE IF NOT EXISTS customer_invoice_payments (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  invoice_id UUID NOT NULL REFERENCES customer_invoices(id) ON DELETE CASCADE,
  payment_date DATE NOT NULL,
  amount DECIMAL(12,2) NOT NULL,
  payment_method VARCHAR(20) CHECK (payment_method IN ('cash', 'mpesa', 'pdq', 'bank_transfer')),
  reference VARCHAR(100),
  recorded_by UUID REFERENCES staff_profiles(id),
  notes TEXT,
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Create aging bucket calculation function
CREATE OR REPLACE FUNCTION calculate_aging_bucket_v2(invoice_date DATE, due_date DATE)
RETURNS VARCHAR(20) AS $$
DECLARE
  days_overdue INTEGER;
BEGIN
  days_overdue := CURRENT_DATE - due_date;
  
  IF days_overdue <= 0 THEN
    RETURN 'current';
  ELSIF days_overdue <= 30 THEN
    RETURN '1-30';
  ELSIF days_overdue <= 60 THEN
    RETURN '31-60';
  ELSIF days_overdue <= 90 THEN
    RETURN '61-90';
  ELSE
    RETURN '90+';
  END IF;
END;
$$ LANGUAGE plpgsql;

-- Trigger to auto-update aging bucket
CREATE OR REPLACE FUNCTION trg_update_aging_bucket_v2()
RETURNS TRIGGER AS $$
BEGIN
  NEW.aging_bucket := calculate_aging_bucket_v2(NEW.invoice_date, NEW.due_date);
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_aging_bucket_customer
BEFORE INSERT OR UPDATE ON customer_invoices
FOR EACH ROW EXECUTE FUNCTION trg_update_aging_bucket_v2();

-- Create update triggers for paid_amount and balance if not already handled by generated columns
-- (Using generated columns for balance, but need to update initial paid_amount on payment insert)

CREATE OR REPLACE FUNCTION update_bill_paid_amount()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_TABLE_NAME = 'employee_credit_payments' THEN
    UPDATE employee_credit_bills 
    SET paid_amount = paid_amount + NEW.amount,
        updated_at = NOW()
    WHERE id = NEW.credit_bill_id;
  ELSIF TG_TABLE_NAME = 'customer_invoice_payments' THEN
    UPDATE customer_invoices 
    SET paid_amount = paid_amount + NEW.amount,
        updated_at = NOW()
    WHERE id = NEW.invoice_id;
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_update_employee_bill_paid
AFTER INSERT ON employee_credit_payments
FOR EACH ROW EXECUTE FUNCTION update_bill_paid_amount();

CREATE TRIGGER trg_update_customer_invoice_paid
AFTER INSERT ON customer_invoice_payments
FOR EACH ROW EXECUTE FUNCTION update_bill_paid_amount();

-- Indexes
CREATE INDEX IF NOT EXISTS idx_employee_credit_employee ON employee_credit_bills(employee_id);
CREATE INDEX IF NOT EXISTS idx_employee_credit_branch ON employee_credit_bills(branch_id);
CREATE INDEX IF NOT EXISTS idx_customer_invoices_customer ON customer_invoices(customer_id);
CREATE INDEX IF NOT EXISTS idx_customer_invoices_aging ON customer_invoices(aging_bucket);
CREATE INDEX IF NOT EXISTS idx_customer_invoices_branch ON customer_invoices(branch_id);
